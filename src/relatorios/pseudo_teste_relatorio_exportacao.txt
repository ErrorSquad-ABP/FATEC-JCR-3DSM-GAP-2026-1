# Pseudocodigo de teste: relatorio + exportacao

Task: T-S2-E3-03
Tipo: Pseudo teste executavel (documental)

Funcao executar_teste_relatorio_exportacao():
  1. cenario_csv()
  2. cenario_pdf()
  3. cenario_consistencia_total()
  4. retornar "cenarios finalizados"

Funcao cenario_csv():
  dado relatorio.total_geral = 120
  quando arquivo = exportar_relatorio(relatorio, "CSV")
  entao assert arquivo.conteudo contem "loja,produto,qty"

Funcao cenario_pdf():
  dado relatorio.total_geral = 120
  quando arquivo = exportar_relatorio(relatorio, "PDF")
  entao assert arquivo.conteudo_simulado contem "Relatorio Consolidado de Estoque"
  entao assert arquivo.conteudo_simulado contem "Total geral"

Funcao cenario_consistencia_total():
  dado relatorio.total_geral = 120
  quando csv = exportar_relatorio(relatorio, "CSV")
  quando pdf = exportar_relatorio(relatorio, "PDF")
  entao assert validar_consistencia_exportacao(relatorio, csv) == "sucesso"
  entao assert validar_consistencia_exportacao(relatorio, pdf) == "sucesso"
