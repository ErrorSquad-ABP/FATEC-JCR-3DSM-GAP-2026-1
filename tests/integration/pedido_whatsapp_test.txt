# Teste integracao pseudocodigo: pedido WhatsApp estoque chat baixa

Task: T-S3-E2E-01
Historia: E2-US01, E2-US02, E2-US03, E3-US01, E3-US02, E4-US01, E4-US02
Fonte: docs/scrum/sprint_backlog/sprint_backlog_ze_mercearia.md

Cenario: fluxo completo de pedido WhatsApp com baixa de estoque e notificacao ao gerente
  Dado que o produto "PROD-001" existe com nome "Produto Teste A"
  E o produto "PROD-001" custa R$ 10,00
  E o produto "PROD-001" possui saldo 15 na "LOJA-01"
  E o cliente se chama "Cliente Teste" e usa o telefone "(11) 99999-0000"
  E existe gerente ativo da "LOJA-01" com WhatsApp configurado
  Quando o cliente enviar "tem Produto Teste A?" pelo WhatsApp
  Entao o chatbot deve responder que "Produto Teste A" custa R$ 10,00
  E deve informar disponibilidade de 15 unidade(s)
  Quando o cliente enviar "quero 2 Produto Teste A"
  Entao o chatbot deve montar um pedido pendente com 2 unidade(s)
  E deve responder um resumo do pedido antes da confirmacao
  Quando o cliente enviar "confirmar"
  Entao o sistema deve confirmar o pedido "PED-001"
  E deve baixar o saldo de "PROD-001" na "LOJA-01" de 15 para 13
  E deve registrar movimentacao de estoque do tipo "SAIDA"
  E deve enviar notificacao WhatsApp ao gerente da "LOJA-01"
  E a notificacao deve conter "Cliente Teste" e "(11) 99999-0000"

Cenario: cancelar pedido pelo chat nao baixa estoque nem notifica gerente
  Dado que o produto "PROD-001" possui saldo 15 na "LOJA-01"
  E o cliente montou um pedido com 2 unidade(s) de "Produto Teste A"
  Quando o cliente enviar "cancelar"
  Entao o pedido deve ficar com status "CANCELADO"
  E o saldo de "PROD-001" na "LOJA-01" deve permanecer 15
  E nao deve existir movimentacao de estoque do tipo "SAIDA" para esse pedido
  E nao deve enviar notificacao WhatsApp ao gerente

Cenario: pedido com estoque insuficiente e recusado antes da confirmacao
  Dado que o produto "PROD-001" possui saldo 15 na "LOJA-01"
  Quando o cliente enviar "quero 100 Produto Teste A"
  Entao o chatbot deve responder que nao ha estoque suficiente
  E nao deve criar pedido confirmado
  E o saldo de "PROD-001" na "LOJA-01" deve permanecer 15
  E nao deve enviar notificacao WhatsApp ao gerente

Cenario: concorrencia basica impede baixa duplicada do mesmo pedido confirmado
  Dado que o pedido "PED-001" ja foi confirmado
  E a baixa de estoque de 2 unidade(s) ja foi registrada para "PED-001"
  E o saldo atual de "PROD-001" na "LOJA-01" e 13
  Quando o sistema receber novamente a mensagem "confirmar" para o mesmo pedido
  Entao o sistema deve ignorar a confirmacao duplicada
  E o saldo de "PROD-001" na "LOJA-01" deve permanecer 13
  E nao deve criar segunda movimentacao de saida para "PED-001"
  E nao deve enviar segunda notificacao ao gerente

Cenario: produto indisponivel sugere alternativa sem baixa de estoque
  Dado que o produto "PROD-002" existe com nome "Produto Zerado"
  E o produto "PROD-002" possui saldo 0 na "LOJA-01"
  E existe o produto alternativo "Produto Teste A" com saldo 15 na "LOJA-01"
  Quando o cliente enviar "quero Produto Zerado"
  Entao o chatbot deve responder que "Produto Zerado" esta indisponivel
  E deve sugerir "Produto Teste A" como alternativa
  E nao deve baixar estoque
  E nao deve notificar o gerente

Pseudo teste executavel:

Funcao executar_teste_pedido_whatsapp_estoque_chat_baixa():
  1. cenario_fluxo_completo_pedido_confirmado()
  2. cenario_cancelamento_nao_baixa_estoque()
  3. cenario_estoque_insuficiente_recusa_pedido()
  4. cenario_confirmacao_duplicada_idempotente()
  5. cenario_produto_indisponivel_sugere_alternativa()
  6. retornar "todos os cenarios executados"

Funcao cenario_fluxo_completo_pedido_confirmado():
  dado produto = criar_produto("PROD-001", "Produto Teste A", 10.00, "ATIVO")
  dado estoque = criar_estoque("PROD-001", "LOJA-01", 15)
  dado cliente = criar_cliente("Cliente Teste", "(11) 99999-0000")
  dado gerente = criar_gerente("LOJA-01", "(11) 98888-0000")
  quando resposta_consulta = webhook_whatsapp_receber(cliente, "tem Produto Teste A?")
  entao assert resposta_consulta.contem("Produto Teste A")
  entao assert resposta_consulta.contem("15 unidade(s)")
  quando resposta_pedido = webhook_whatsapp_receber(cliente, "quero 2 Produto Teste A")
  entao assert resposta_pedido.contem("resumo do pedido")
  quando resposta_confirmacao = webhook_whatsapp_receber(cliente, "confirmar")
  entao assert pedido("PED-001").status == "CONFIRMADO"
  entao assert estoque_saldo("PROD-001", "LOJA-01") == 13
  entao assert movimentacao_existe("PED-001", "SAIDA")
  entao assert notificacao_gerente_existe("PED-001", "ENVIADA")
  entao assert notificacao_gerente("PED-001").mensagem.contem("Cliente Teste")

Funcao cenario_cancelamento_nao_baixa_estoque():
  dado estoque = criar_estoque("PROD-001", "LOJA-01", 15)
  dado pedido = criar_pedido_pendente("PED-002", "PROD-001", 2)
  quando resposta = webhook_whatsapp_receber(pedido.cliente, "cancelar")
  entao assert pedido.status == "CANCELADO"
  entao assert estoque_saldo("PROD-001", "LOJA-01") == 15
  entao assert nao_existe_movimentacao("PED-002", "SAIDA")
  entao assert nao_existe_notificacao_gerente("PED-002")

Funcao cenario_estoque_insuficiente_recusa_pedido():
  dado estoque = criar_estoque("PROD-001", "LOJA-01", 15)
  quando resposta = webhook_whatsapp_receber(cliente_padrao(), "quero 100 Produto Teste A")
  entao assert resposta.contem("nao ha estoque suficiente")
  entao assert repositorio_pedidos.nao_existe_confirmado_para("PROD-001")
  entao assert estoque_saldo("PROD-001", "LOJA-01") == 15
  entao assert repositorio_notificacoes.total() == 0

Funcao cenario_confirmacao_duplicada_idempotente():
  dado pedido = criar_pedido_confirmado("PED-001", "PROD-001", 2)
  dado movimentacao = criar_movimentacao_saida("PED-001", "PROD-001", 2)
  dado estoque = criar_estoque("PROD-001", "LOJA-01", 13)
  quando resposta = webhook_whatsapp_receber(pedido.cliente, "confirmar")
  entao assert resposta.contem("pedido ja confirmado")
  entao assert estoque_saldo("PROD-001", "LOJA-01") == 13
  entao assert contar_movimentacoes("PED-001", "SAIDA") == 1
  entao assert contar_notificacoes("PED-001") == 1

Funcao cenario_produto_indisponivel_sugere_alternativa():
  dado produto_zerado = criar_produto("PROD-002", "Produto Zerado", 8.00, "ATIVO")
  dado estoque_zerado = criar_estoque("PROD-002", "LOJA-01", 0)
  dado alternativa = criar_produto("PROD-001", "Produto Teste A", 10.00, "ATIVO")
  dado estoque_alternativa = criar_estoque("PROD-001", "LOJA-01", 15)
  quando resposta = webhook_whatsapp_receber(cliente_padrao(), "quero Produto Zerado")
  entao assert resposta.contem("Produto Zerado")
  entao assert resposta.contem("indisponivel")
  entao assert resposta.contem("Produto Teste A")
  entao assert repositorio_movimentacoes.total() == 0
  entao assert repositorio_notificacoes.total() == 0
