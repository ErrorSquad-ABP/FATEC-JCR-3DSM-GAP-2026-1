# Teste pseudocodigo: notificacao WhatsApp ao gerente

Task: T-S3-C4-03
Historia: E4-US01, E4-US02
Fonte: docs/scrum/sprint_backlog/sprint_backlog_ze_mercearia.md

Cenario: disparar notificacao apenas apos confirmacao do pedido
  Dado que o pedido "PED-001" existe com status "CONFIRMADO"
  E a baixa de estoque do pedido "PED-001" foi concluida com sucesso
  E existe gerente ativo vinculado a "LOJA-01" com telefone WhatsApp configurado
  Quando o sistema processar a confirmacao do pedido "PED-001"
  Entao deve enviar uma notificacao WhatsApp ao gerente da "LOJA-01"
  E deve registrar a notificacao com status "ENVIADA"

Cenario: nao disparar notificacao para pedido ainda nao confirmado
  Dado que o pedido "PED-002" existe com status "EM_RESUMO"
  E existe gerente ativo vinculado a "LOJA-01"
  Quando o sistema tentar notificar o gerente
  Entao deve retornar erro "pedido nao confirmado"
  E nao deve chamar o gateway WhatsApp
  E nao deve registrar notificacao com status "ENVIADA"

Cenario: nao disparar notificacao quando pedido for cancelado
  Dado que o pedido "PED-003" existe com status "CANCELADO"
  Quando o cliente cancelar o pedido pelo chat
  Entao o sistema nao deve enviar notificacao WhatsApp ao gerente
  E deve retornar erro "pedido nao confirmado"

Cenario: notificar gerente com resumo do pedido e dados do cliente
  Dado que o pedido "PED-001" esta confirmado na "LOJA-01"
  E o cliente do pedido se chama "Cliente Teste"
  E o telefone do cliente e "(11) 99999-0000"
  E o pedido possui 2 unidade(s) de "Produto Teste A"
  Quando a notificacao for enviada ao gerente
  Entao a mensagem deve conter "Pedido confirmado: PED-001"
  E deve conter "Cliente: Cliente Teste"
  E deve conter "Telefone: (11) 99999-0000"
  E deve conter "2x Produto Teste A"

Cenario: rejeitar notificacao quando gerente da loja nao existe
  Dado que o pedido "PED-004" esta confirmado na "LOJA-99"
  E nao existe gerente ativo vinculado a "LOJA-99"
  Quando o sistema tentar notificar o gerente
  Entao deve retornar erro "gerente da loja nao encontrado"
  E nao deve chamar o gateway WhatsApp

Cenario: rejeitar notificacao quando telefone do gerente nao esta configurado
  Dado que o pedido "PED-005" esta confirmado na "LOJA-01"
  E o gerente da "LOJA-01" nao possui telefone WhatsApp configurado
  Quando o sistema tentar notificar o gerente
  Entao deve retornar erro "telefone do gerente nao configurado"
  E nao deve chamar o gateway WhatsApp

Cenario: registrar falha quando gateway WhatsApp nao enviar mensagem
  Dado que o pedido "PED-006" esta confirmado na "LOJA-01"
  E existe gerente ativo com telefone WhatsApp configurado
  E o gateway WhatsApp retorna falha de envio
  Quando o sistema tentar enviar a notificacao
  Entao deve retornar erro "falha ao enviar notificacao"
  E deve registrar a notificacao com status "ERRO_ENVIO"

Pseudo teste executavel:

Funcao executar_teste_notificacao_whatsapp_gerente():
  1. cenario_notificar_apos_confirmacao()
  2. cenario_nao_notificar_pedido_nao_confirmado()
  3. cenario_nao_notificar_pedido_cancelado()
  4. cenario_mensagem_contem_resumo_cliente()
  5. cenario_gerente_nao_encontrado()
  6. cenario_telefone_gerente_nao_configurado()
  7. cenario_falha_gateway_registrada()
  8. retornar "todos os cenarios executados"

Funcao cenario_notificar_apos_confirmacao():
  dado pedido = criar_pedido("PED-001", "LOJA-01", "CONFIRMADO")
  dado baixa_estoque = criar_baixa_concluida("PED-001")
  dado gerente = criar_gerente("GER-001", "LOJA-01", "(11) 98888-0000")
  quando resultado = notificar_gerente_pedido_confirmado(pedido)
  entao assert resultado == "sucesso"
  entao assert gateway_whatsapp.foi_chamado_com("(11) 98888-0000")
  entao assert repositorio_notificacoes.existe("PED-001", "ENVIADA")

Funcao cenario_nao_notificar_pedido_nao_confirmado():
  dado pedido = criar_pedido("PED-002", "LOJA-01", "EM_RESUMO")
  quando resultado = notificar_gerente_pedido_confirmado(pedido)
  entao assert resultado.erro == "pedido nao confirmado"
  entao assert gateway_whatsapp.total_chamadas() == 0
  entao assert repositorio_notificacoes.nao_existe_status("PED-002", "ENVIADA")

Funcao cenario_nao_notificar_pedido_cancelado():
  dado pedido = criar_pedido("PED-003", "LOJA-01", "CANCELADO")
  quando resultado = notificar_gerente_pedido_confirmado(pedido)
  entao assert resultado.erro == "pedido nao confirmado"
  entao assert gateway_whatsapp.total_chamadas() == 0

Funcao cenario_mensagem_contem_resumo_cliente():
  dado pedido = criar_pedido_confirmado_com_cliente("PED-001", "LOJA-01", "Cliente Teste", "(11) 99999-0000")
  dado pedido.itens = [ { produto_nome: "Produto Teste A", quantidade: 2, preco_unitario: 10.00 } ]
  dado gerente = criar_gerente("GER-001", "LOJA-01", "(11) 98888-0000")
  quando resultado = notificar_gerente_pedido_confirmado(pedido)
  entao assert resultado == "sucesso"
  entao assert gateway_whatsapp.ultima_mensagem().contem("Pedido confirmado: PED-001")
  entao assert gateway_whatsapp.ultima_mensagem().contem("Cliente: Cliente Teste")
  entao assert gateway_whatsapp.ultima_mensagem().contem("Telefone: (11) 99999-0000")
  entao assert gateway_whatsapp.ultima_mensagem().contem("2x Produto Teste A")

Funcao cenario_gerente_nao_encontrado():
  dado pedido = criar_pedido("PED-004", "LOJA-99", "CONFIRMADO")
  dado repositorio_usuarios.sem_gerente_ativo("LOJA-99")
  quando resultado = notificar_gerente_pedido_confirmado(pedido)
  entao assert resultado.erro == "gerente da loja nao encontrado"
  entao assert gateway_whatsapp.total_chamadas() == 0

Funcao cenario_telefone_gerente_nao_configurado():
  dado pedido = criar_pedido("PED-005", "LOJA-01", "CONFIRMADO")
  dado gerente = criar_gerente("GER-001", "LOJA-01", vazio)
  quando resultado = notificar_gerente_pedido_confirmado(pedido)
  entao assert resultado.erro == "telefone do gerente nao configurado"
  entao assert gateway_whatsapp.total_chamadas() == 0

Funcao cenario_falha_gateway_registrada():
  dado pedido = criar_pedido("PED-006", "LOJA-01", "CONFIRMADO")
  dado gerente = criar_gerente("GER-001", "LOJA-01", "(11) 98888-0000")
  dado gateway_whatsapp.configurar_retorno("erro")
  quando resultado = notificar_gerente_pedido_confirmado(pedido)
  entao assert resultado.erro == "falha ao enviar notificacao"
  entao assert repositorio_notificacoes.existe("PED-006", "ERRO_ENVIO")
